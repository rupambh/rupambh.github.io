(function() {
  function initTalkFilters() {
    var state = {
      cat: 'invited',
      status: 'all',
      forum: 'all',
      paper: 'all',
      search: ''
    };

    var items = document.querySelectorAll('.talk-entry');
    var countHeader = document.getElementById('talksCountHeader');
    var searchInput = document.getElementById('searchTalksInput');

    function update() {
      var visible = 0;
      var q = state.search.trim().toLowerCase();

      items.forEach(function(el) {
        var c = el.getAttribute('data-cat');
        var st = el.getAttribute('data-status');
        var ha = el.getAttribute('data-has-award');
        var f = el.getAttribute('data-forum');
        var p = el.getAttribute('data-paper');
        var s = el.getAttribute('data-search');

        var matchCat = (state.cat === 'all' || c === state.cat);
        var matchForum = (state.forum === 'all' || f === state.forum);
        var matchPaper = (state.paper === 'all' || p === state.paper);
        
        var matchStatus = true;
        if (state.status === 'presented') {
          matchStatus = (st === 'presented');
        } else if (state.status === 'selected') {
          matchStatus = (st === 'selected');
        } else if (state.status === 'awards_only') {
          matchStatus = (ha === 'true');
        }

        var matchSearch = (!q || (s && s.indexOf(q) !== -1));

        if (matchCat && matchStatus && matchForum && matchPaper && matchSearch) {
          el.style.display = 'block';
          visible++;
        } else {
          el.style.display = 'none';
        }
      });

      if (countHeader) {
        if (state.cat === 'all' && state.status === 'all' && state.forum === 'all' && state.paper === 'all' && !q) {
          countHeader.textContent = 'Showing all ' + items.length + ' presentations';
        } else {
          countHeader.textContent = 'Showing ' + visible + ' of ' + items.length + ' presentations';
        }
      }
    }

    document.querySelectorAll('.filter-btn').forEach(function(btn) {
      btn.addEventListener('click', function(e) {
        e.preventDefault();
        var type = this.getAttribute('data-filter-type');
        var val = this.getAttribute('data-filter-val');

        var group = this.parentElement;
        group.querySelectorAll('.filter-btn').forEach(function(b) {
          b.classList.remove('active');
        });
        this.classList.add('active');

        state[type] = val;
        update();
      });
    });

    if (searchInput) {
      searchInput.addEventListener('input', function() {
        state.search = this.value;
        update();
      });
    }

    // Initialize default filtered view on load (Invited)
    update();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initTalkFilters);
  } else {
    initTalkFilters();
  }
})();
</script>
