<script>
(function() {
  function initServiceFilters() {
    var state = {
      role: 'all',
      domain: 'all',
      search: ''
    };

    var items = document.querySelectorAll('.service-entry');
    var countHeader = document.getElementById('serviceCountHeader');
    var searchInput = document.getElementById('searchServiceInput');

    function update() {
      var visible = 0;
      var q = state.search.trim().toLowerCase();

      items.forEach(function(el) {
        var r = el.getAttribute('data-role');
        var d = el.getAttribute('data-domain');
        var s = el.getAttribute('data-search');

        var matchRole = (state.role === 'all' || r === state.role);
        var matchDomain = (state.domain === 'all' || d === state.domain);
        var matchSearch = (!q || (s && s.indexOf(q) !== -1));

        if (matchRole && matchDomain && matchSearch) {
          el.style.display = 'block';
          visible++;
        } else {
          el.style.display = 'none';
        }
      });

      if (countHeader) {
        if (state.role === 'all' && state.domain === 'all' && !q) {
          countHeader.textContent = 'Showing all ' + items.length + ' activities';
        } else {
          countHeader.textContent = 'Showing ' + visible + ' of ' + items.length + ' activities';
        }
      }
    }

    document.querySelectorAll('.filter-btn').forEach(function(btn) {
      btn.addEventListener('click', function(e) {
        e.preventDefault();
        var type = this.getAttribute('data-filter-type');
        var val = this.getAttribute('data-filter-val');

        var group = this.parentElement;
        group.querySelectorAll('.filter-btn').forEach(function(b) {
          b.classList.remove('active');
        });
        this.classList.add('active');

        state[type] = val;
        update();
      });
    });

    if (searchInput) {
      searchInput.addEventListener('input', function() {
        state.search = this.value;
        update();
      });
    }

    update();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initServiceFilters);
  } else {
    initServiceFilters();
  }
})();
</script>
