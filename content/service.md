(function() {
  function initJournalFilters() {
    var state = {
      domain: 'methodology',
      search: ''
    };

    var items = document.querySelectorAll('.journal-entry');
    var countHeader = document.getElementById('journalCountHeader');
    var searchInput = document.getElementById('searchJournalInput');

    function update() {
      var visible = 0;
      var q = state.search.trim().toLowerCase();

      items.forEach(function(el) {
        var d = el.getAttribute('data-domain');
        var s = el.getAttribute('data-search');

        var matchDomain = (state.domain === 'all' || d === state.domain);
        var matchSearch = (!q || (s && s.indexOf(q) !== -1));

        if (matchDomain && matchSearch) {
          el.style.display = 'block';
          visible++;
        } else {
          el.style.display = 'none';
        }
      });

      if (countHeader) {
        if (state.domain === 'all' && !q) {
          countHeader.textContent = 'Showing all ' + items.length + ' journals';
        } else {
          countHeader.textContent = 'Showing ' + visible + ' of ' + items.length + ' journals';
        }
      }
    }

    document.querySelectorAll('.filter-btn').forEach(function(btn) {
      btn.addEventListener('click', function(e) {
        e.preventDefault();
        var val = this.getAttribute('data-domain');

        var group = this.parentElement;
        group.querySelectorAll('.filter-btn').forEach(function(b) {
          b.classList.remove('active');
        });
        this.classList.add('active');

        state.domain = val;
        update();
      });
    });

    if (searchInput) {
      searchInput.addEventListener('input', function() {
        state.search = this.value;
        update();
      });
    }

    update();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initJournalFilters);
  } else {
    initJournalFilters();
  }
})();
</script>
